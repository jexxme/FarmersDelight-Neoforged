export const chestMods = modifiers.filter((m) => m.type === 'farmersdelight:add_loot_table');
export const tables = chestMods.map((m) => m.table).filter((t) => loot.chests[t]);
export const allItems = [...new Set(tables.flatMap((t) => loot.chests[t].pools.flatMap((p) => p.entries.map((e) => e.item).filter(Boolean))))];
export const where = allItems.map((id) => ({ id, in: tables.map((t) => ({ t, p: chestChance(loot.chests[t], id).chance })).filter((x) => x.p > 0).sort((a, b) => b.p - a.p) }));
export const groups = [
	['Mineshafts, dungeons and shipwrecks', /mineshaft|dungeon|shipwreck/, 'Seeds, rice and plenty of rope, plus the odd Cooking Pot or Skillet.'],
	['Pillager outposts', /outpost/, 'Food for the raiders.'],

].map(([title, re, text]) => ({ title, text, list: tables.filter((t) => re.test(t)) })).filter((g) => g.list.length);
export const other = tables.filter((t) => !groups.some((g) => g.list.includes(t)));
export const bsLoot = tags.resolved.enchantment['minecraft:on_random_loot']?.includes('farmersdelight:backstabbing');

Farmer's Delight adds its own loot to {tables.length} kinds of chest: seeds and rope in mineshafts and shipwrecks, knives in bastions and end cities, onions in pillager outposts, and crops in village houses. The normal loot stays; these items come on top. It also gives some mobs an extra drop.

## Mob drops

Zombies and illagers killed by a player can drop an <Item id="onion" />:

<LootModifiers ids={['farmersdelight:add_onion_to_zombies', 'farmersdelight:add_onion_to_illagers']} />

Killing animals with a [knife](/mechanics/knives/#extra-drops-with-a-knife) gives extra leather, ham, feathers and more, and knives get straw from grass and slices from cakes and pies. Those drops are listed on the [Knives](/mechanics/knives/) and [Straw](/mechanics/straw-and-rope/#getting-straw) pages.

## Good to know

- Chests roll their loot the first time they're opened. Chests that were already opened keep what they had.
- Server owners can turn the chest loot off in the [config](/reference/config/). The mob drops always stay on.
