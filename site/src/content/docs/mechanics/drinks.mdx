export const drinks = itemList.filter((i) => i.consumable && (i.category === 'drinks' || i.consumable.animation === 'drink'));
export const tab = drinks.filter((d) => d.category === 'drinks');
export const noFood = drinks.filter((i) => !i.food && i.consumable.other.length);
export const describe = (o) =>
	o.type === 'farmersdelight:remove_random_effects'
		? `Clears one random ${o.harmful_only ? 'harmful ' : ''}effect`
		: o.type === 'farmersdelight:heal'
			? `Heals ${hearts(o.amount)}`
			: o.type === 'farmersdelight:extinguish'
				? 'Puts out fire'
				: o.type;
export const stacks = [...new Set(drinks.map((d) => d.stack))];
export const alwaysOk = drinks.filter((d) => !d.food || d.food.alwaysEdible);

## Remedies

{noFood.map((d, i) => <>{i ? (i === noFood.length - 1 ? ' and ' : ', ') : ''}<Item id={d.id} /></>)} don't fill your
hunger at all. They're remedies, and you can drink them whenever you like, even on a full hunger bar:

<ul>
	{noFood.map((d) => <li><Item id={d.id} />: {d.consumable.other.map((o) => describe(o).toLowerCase()).join(', ')}.</li>)}
</ul>

A Milk Bottle is a bite-sized milk bucket: one bottle takes away a single effect instead of all of them, so you can shake
off Poison without losing your other buffs (if you're lucky with the pick). Hot Cocoa only ever touches harmful
effects, so it never costs you a good one.

## Good to know

- **Drink when full.** {alwaysOk.map((d, i) => <>{i ? (i === alwaysOk.length - 1 ? ' and ' : ', ') : ''}<Item id={d.id} /></>)} can be drunk even with a full hunger bar.
- **The bottle comes back** even when you drink from a stack: it goes into your inventory, or drops at your feet if
  that's full.
- **Clearing effects is random.** A drink that clears one effect picks any of your active effects at random (or any
  harmful one), so it may not hit the one you wanted.
