export const feasts = tags.resolved.block['farmersdelight:feasts'];
export const maxOf = (id) => Math.max(...blocks[id].properties.servings.map(Number));
export const servingOf = (id) => id.replace(/_block$/, '');
export const lootOf = (id) => {
	const t = loot.blocks[id.replace('farmersdelight:', 'farmersdelight:blocks/')];
	const max = maxOf(id);
	const full = t.pools.filter((p) => p.conditions.includes(`servings = ${max}`)).flatMap((p) => p.entries.map((e) => e.item));
	const part = t.pools.filter((p) => p.conditions.includes(`not (servings = ${max})`)).flatMap((p) => p.entries.map((e) => e.item));
	return { full, part };
};
export const medley = 'farmersdelight:rice_roll_medley_block';
export const fourServing = feasts.filter((id) => id !== medley);
export const noLeftovers = feasts.filter((id) => !feast.registrations.value[id]);
export const salad = 'farmersdelight:gleaming_salad_block';
export const servingKey = (id) => { const s = items[servingOf(id)]; return JSON.stringify([s.food, s.consumable.effects]); };
export const sameServing = new Set(fourServing.map(servingKey)).size === 1;
export const firstServing = items[servingOf(fourServing[0])];
export const ladderLabels = (id) => Object.fromEntries(Array.from({ length: maxOf(id) + 1 }, (_, n) => [n, n === maxOf(id) ? 'full' : n === 0 ? (feast.registrations.value[id] ? 'leftovers' : 'gone') : `${n} left`]));
export const medleyRuns = [...feast.medleyOrder.value].reverse().reduce((runs, id) => {
	const last = runs[runs.length - 1];
	if (last && last.id === id) last.n++;
	else runs.push({ id, n: 1 });
	return runs;
}, []);

Feasts are big meals you place as a block and share. Each one holds several servings, and anyone can take one by
right-clicking it with the right container, usually a <Item id="minecraft:bowl" />. There are {feasts.length} feasts.
You cook or craft each one as a single item that stacks to {items['farmersdelight:roast_chicken_block'].stack}.

## Serving

Place the feast on a solid block, then right-click it holding the serving's container. You use up one container and get
one serving, and the feast shows one serving less. Holding the wrong thing shows a hint that names the container.

<ServingLadder block="roast_chicken_block" />

When the last serving is gone, most feasts leave a plate of leftovers. Right-click it to clear it away and get back what
is left, such as a bone or a bowl. {noLeftovers.map((id) => <Item id={id} />)} {noLeftovers.length === 1 ? 'is the exception: it disappears' : 'are the exceptions: they disappear'} with the last serving.

## Breaking a feast

Break a **full** feast and you get it back as an item, ready to place again. Break it after even one serving and the
servings left on it are lost: you only get the leftovers.

<table>
	<thead><tr><th>Feast</th><th>Broken while full</th><th>Broken after serving</th></tr></thead>
	<tbody>
		{feasts.map((id) => {
			const l = lootOf(id);
			return (
				<tr>
					<td><Item id={id} /></td>
					<td>{l.full.map((x, i) => <>{i ? ', ' : ''}<Item id={x} /></>)}</td>
					<td>{l.part.length ? l.part.map((x, i) => <>{i ? ', ' : ''}<Item id={x} /></>) : <em>nothing</em>}</td>
				</tr>
			);
		})}
	</tbody>
</table>

<Aside type="caution">
Serve every portion before you break a feast. A roast chicken with three servings left breaks into leftovers, not
three meals.
</Aside>

## Rice Roll Medley

The <Item id={medley} /> holds {feast.medleyServings.value} rolls, served in order:
{medleyRuns.map((r, i) => <>{i ? (i === medleyRuns.length - 1 ? ', then ' : ', then ') : ''}{r.n} <Item id={r.id} plural /></>)}.

<Aside type="caution" title="Known issue in 1.3.4">
The medley can't be served: right-clicking it does nothing. Break it while it's still full to get it back, and make the
rolls on their own instead.
</Aside>

## Good to know

- **Gleaming Salad glows.** The <Item id={salad} /> gives off light level {feast.saladLightPerServing.value} per serving
  left, so {blocks[salad].maxLight} when full, and it sparkles.
- **Comparators** next to a feast read how many servings are left.
- **Pistons** break a feast instead of pushing it.
- **Mobs** walk around feasts instead of over them.
